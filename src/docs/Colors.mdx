export const ColorSwatch = ({ label, value, showLabel = true }) => {
  return (
    <div>
      {value ? (
        <button
          type="button"
          title={value}
          style={{ backgroundColor: value }}
          className="aspect-square w-full rounded-sm outline -outline-offset-1 outline-black/10"
        ></button>
      ) : (
        <div className="aspect-square w-full"></div>
      )}
      {showLabel && (
        <span className="block text-center text-[10px] text-gray-500 mt-1 ">{label}</span>
      )}
    </div>
  );
};

export const ColorGridSection = ({ title, data}) => (
  <div className="space-y-4">
    <h3 className="text-xl font-bold text-gray-900 mb-4">{title}</h3>
    <div className="not-prose space-y-4">
      {(
        Object.entries(data).map(([groupName, values]) => {
          const entries = Object.entries(values);
          return (
            <div key={groupName} className="grid grid-cols-[auto_minmax(0,1fr)] items-center gap-4">
              <p className="font-medium text-gray-950 sm:pr-12 w-28">{groupName}</p>
              <div className="grid gap-1.5 sm:gap-4 grid-flow-col auto-cols-fr">
                {entries.map(([key, val]) => (
                  <ColorSwatch key={key} label={key} value={val} />
                ))}
              </div>
            </div>
          );
        })
      )}
    </div>
  </div>
);
